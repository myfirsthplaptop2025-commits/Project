    Source Code — task state & persistence
    function saveTasks() {
  try {
    localStorage.setItem(
      "shahriar-taskflow-v2",
      JSON.stringify(tasks)
    );
  } catch (error) {
    showStorageWarning(error);
  }
}

function completeTask(id) {
  const task = tasks.find(item => item.id === id);
  if (!task) return;
  task.done = !task.done;
  task.completedAt = task.done ? localDate() : "";
  saveTasks();
  render();
}
