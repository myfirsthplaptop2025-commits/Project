So we brew with care, bake in small batches, and always leave room at the table. The coffee brings you in. The feeling makes it yours.

Source Code — menu filters & basket 
function filterMenu() {
  const query = search.value.trim().toLowerCase();
  items.forEach(item => {
    const matchesCategory =
      category === "all" ||
      item.dataset.category === category;
    const matchesSearch =
      item.dataset.search.includes(query);
    item.hidden = !(matchesCategory && matchesSearch);
  });
}

function addToBasket(item) {
  const current = basket.get(item.id);
  basket.set(item.id, {
    ...item,
    quantity: current ? current.quantity + 1 : 1
  });
  renderBasket();
}
